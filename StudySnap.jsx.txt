// ─── Anthropic API call ───────────────────────────────────────────────────────
async function callClaude(systemPrompt, userPrompt) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
      system: systemPrompt,
      messages: [{ role: "user", content: userPrompt }],
    }),
  });
  const data = await res.json();
  return data.content?.[0]?.text || "";
}

async function callClaudeJSON(systemPrompt, userPrompt) {
  const text = await callClaude(systemPrompt + "\n\nIMPORTANT: Respond ONLY with valid JSON. No markdown, no backticks, no preamble.", userPrompt);
  try { return JSON.parse(text.replace(/```json|```/g, "").trim()); }
  catch { return null; }
}

// ─── Auth Context (localStorage mock) ────────────────────────────────────────
function useAuth() {
  const [user, setUser] = useState(() => {
    try { return JSON.parse(localStorage.getItem("ss_user") || "null"); } catch { return null; }
  });

  const signup = (name, email, password) => {
    const u = { id: crypto.randomUUID(), name, email, plan: "free", streak: 1, joinDate: new Date().toISOString(), dailyUsage: 0, lastUsageDate: "" };
    localStorage.setItem("ss_user", JSON.stringify(u));
    localStorage.setItem(`ss_pw_${email}`, password);
    setUser(u);
    return u;
  };

  const login = (email, password) => {
    const stored = localStorage.getItem(`ss_pw_${email}`);
    if (stored !== password) throw new Error("Invalid credentials");
    const u = JSON.parse(localStorage.getItem("ss_user") || "{}");
    if (u.email !== email) throw new Error("User not found");
    const today = new Date().toDateString();
    if (u.lastUsageDate !== today) { u.lastUsageDate = today; u.dailyUsage = 0; }
    setUser(u);
    return u;
  };

  const logout = () => { setUser(null); };

  const updateUser = (updates) => {
    const u = { ...user, ...updates };
    localStorage.setItem("ss_user", JSON.stringify(u));
    setUser(u);
  };

  const canUseAI = () => {
    if (!user) return false;
    if (user.plan === "premium") return true;
    const today = new Date().toDateString();
    if (user.lastUsageDate !== today) return true;
    return (user.dailyUsage || 0) < 5;
  };

  const recordUsage = () => {
    const today = new Date().toDateString();
    const usage = user.lastUsageDate === today ? (user.dailyUsage || 0) + 1 : 1;
    updateUser({ dailyUsage: usage, lastUsageDate: today });
  };

  return { user, signup, login, logout, updateUser, canUseAI, recordUsage };
}

// ─── Storage Helpers ──────────────────────────────────────────────────────────
function getSavedNotes(userId) {
  try { return JSON.parse(localStorage.getItem(`ss_notes_${userId}`) || "[]"); } catch { return []; }
}
function saveNote(userId, note) {
  const notes = getSavedNotes(userId);
  const idx = notes.findIndex(n => n.id === note.id);
  if (idx >= 0) notes[idx] = note; else notes.unshift(note);
  localStorage.setItem(`ss_notes_${userId}`, JSON.stringify(notes.slice(0, 50)));
}
function deleteNote(userId, noteId) {
  const notes = getSavedNotes(userId).filter(n => n.id !== noteId);
  localStorage.setItem(`ss_notes_${userId}`, JSON.stringify(notes));
}

// ─── UPLOAD / STUDY PAGE ──────────────────────────────────────────────────────
function UploadPage({ user, onProcess, canUse, setPage }) {
  const [text, setText] = useState("");
  const [subject, setSubject] = useState("other");
  const [title, setTitle] = useState("");
  const [dragging, setDragging] = useState(false);
  const [loading, setLoading] = useState(false);
  const fileRef = useRef(null);

  const handleFile = async (file) => {
    if (!file) return;
    if (file.type === "text/plain") {
      const t = await file.text();
      setText(t);
    } else if (file.type === "application/pdf") {
      if (user?.plan !== "premium") { alert("PDF uploads are a Premium feature. Please upgrade!"); setPage("pricing"); return; }
      setText("[PDF uploaded — OCR processing would extract text here in production]");
    } else if (file.type.startsWith("image/")) {
      if (user?.plan !== "premium") { alert("Image OCR is a Premium feature. Please upgrade!"); setPage("pricing"); return; }
      setText("[Image uploaded — OCR would extract handwritten text here in production]");
    }
  };

  const handleDrop = (e) => {
    e.preventDefault(); setDragging(false);
    handleFile(e.dataTransfer.files[0]);
  };

  const handleProcess = async () => {
    if (!text.trim()) return;
    if (!canUse()) { alert("You've used all 5 free summaries today. Upgrade to Premium for unlimited!"); setPage("pricing"); return; }
    setLoading(true);
    try {
      await onProcess(text, title || "My Notes", subject);
    } finally { setLoading(false); }
  };

  return (
    <div className="slide-up" style={{ maxWidth: 760, margin: "0 auto", padding: "28px 24px", display: "flex", flexDirection: "column", gap: 20 }}>
      <div>
        <h2 style={{ fontWeight: 800, fontSize: 22, marginBottom: 4 }}>⚡ New Study Session</h2>
        <p style={{ color: "var(--text2)", fontSize: 14 }}>Paste or upload your notes and let AI do the heavy lifting.</p>
      </div>

      <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 14 }}>
        <div>
          <label style={{ fontSize: 12, fontWeight: 600, color: "var(--text2)", display: "block", marginBottom: 6 }}>Title</label>
          <input placeholder="e.g. Photosynthesis Ch.4" value={title} onChange={e => setTitle(e.target.value)} />
        </div>
        <div>
          <label style={{ fontSize: 12, fontWeight: 600, color: "var(--text2)", display: "block", marginBottom: 6 }}>Subject</label>
          <select value={subject} onChange={e => setSubject(e.target.value)}>
            {SUBJECTS.map(s => <option key={s.id} value={s.id}>{s.icon} {s.label}</option>)}
          </select>
        </div>
      </div>

      {/* Drop zone */}
      <div
        onDragOver={e => { e.preventDefault(); setDragging(true); }}
        onDragLeave={() => setDragging(false)}
        onDrop={handleDrop}
        onClick={() => fileRef.current?.click()}
        style={{ border: `2px dashed ${dragging ? "var(--accent)" : "var(--border)"}`, borderRadius: 14, padding: 24, textAlign: "center", cursor: "pointer", background: dragging ? "var(--accent-light)" : "var(--surface2)", transition: "all .2s" }}>
        <input ref={fileRef} type="file" style={{ display: "none" }} accept=".txt,.pdf,image/*" onChange={e => handleFile(e.target.files[0])} />
        <div style={{ fontSize: 32, marginBottom: 8 }}>📎</div>
        <div style={{ fontWeight: 600, fontSize: 14 }}>Drop a file here or click to upload</div>
        <div style={{ fontSize: 12, color: "var(--text3)", marginTop: 4 }}>
          .txt free · PDF & images {user?.plan === "premium" ? "✓ included" : "Premium only"}
        </div>
      </div>

      <div style={{ textAlign: "center", color: "var(--text3)", fontSize: 13, fontWeight: 600 }}>— or paste your notes below —</div>

      <textarea
        value={text}
        onChange={e => setText(e.target.value)}
        placeholder="Paste your lecture notes, textbook content, or any study material here..."
        style={{ minHeight: 220, resize: "vertical", lineHeight: 1.7, fontSize: 14 }}
      />

      <div style={{ display: "flex", gap: 12, justifyContent: "flex-end", flexWrap: "wrap" }}>
        <div style={{ fontSize: 12, color: "var(--text3)", alignSelf: "center" }}>{text.length} characters</div>
        <button className="btn btn-ghost" onClick={() => setText("")} disabled={!text}>Clear</button>
        <button className="btn btn-gradient" style={{ padding: "12px 28px", fontSize: 15 }} onClick={handleProcess} disabled={loading || !text.trim()}>
          {loading ? <><span className="loader" style={{ width: 18, height: 18 }} /> Processing…</> : "⚡ Generate Study Pack"}
        </button>
      </div>

      {!canUse() && (
        <div style={{ background: "#fee2e2", borderRadius: 12, padding: 16, display: "flex", justifyContent: "space-between", alignItems: "center" }}>
          <div style={{ fontSize: 13, color: "var(--red)", fontWeight: 600 }}>⚠ Daily limit reached (5/5)</div>
          <button className="btn btn-gradient" style={{ padding: "8px 16px", fontSize: 12 }} onClick={() => setPage("pricing")}>Upgrade →</button>
        </div>
      )}
    </div>
  );
}

  const [flipped, setFlipped] = useState(false);
  const [loading, setLoading] = useState(false);
  const [known, setKnown] = useState(new Set());
  const [mode, setMode] = useState("browse"); // browse | study

  const generate = useCallback(async (n) => {
    if (!n?.text) return;
    setLoading(true);
    try {
      const result = await callClaudeJSON(
        "You are a flashcard generator. Return JSON only.",
        `Create 8 flashcard pairs from these notes. Return: {"cards": [{"front": "question/term", "back": "answer/definition"},...]}
        
Notes: ${n.text.slice(0, 2000)}`
      );
      if (result?.cards) setCards(result.cards);
    } catch (e) {
      setCards([
        { front: "What is the main topic?", back: n.title || "See your notes" },
        { front: "Key concept #1", back: "Review your uploaded content" },
      ]);
    } finally { setLoading(false); }
  }, []);

  useEffect(() => {
    if (note) generate(note);
    else {
      const saved = getSavedNotes(user?.id || "");
      if (saved[0]) generate(saved[0]);
    }
  }, [note, user, generate]);

  const next = () => { setCurrent(c => Math.min(c + 1, cards.length - 1)); setFlipped(false); };
  const prev = () => { setCurrent(c => Math.max(c - 1, 0)); setFlipped(false); };
  const markKnown = () => { setKnown(k => new Set([...k, current])); next(); };

  if (!note && !getSavedNotes(user?.id || "").length) return (
    <div style={{ padding: 40, textAlign: "center", color: "var(--text2)" }}>
      <div style={{ fontSize: 48, marginBottom: 16 }}>🃏</div>
      <div style={{ fontWeight: 600 }}>No flashcards yet</div>
      <button className="btn btn-primary" style={{ marginTop: 20 }} onClick={() => setPage("upload")}>Upload notes first</button>
    </div>
  );

  return (
    <div className="slide-up" style={{ maxWidth: 680, margin: "0 auto", padding: "28px 24px", display: "flex", flexDirection: "column", gap: 20 }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
        <div>
          <h2 style={{ fontWeight: 800, fontSize: 22, marginBottom: 4 }}>🃏 Flashcards</h2>
          <p style={{ color: "var(--text2)", fontSize: 13 }}>{note?.title || "Study Deck"} · {cards.length} cards</p>
        </div>
        <div style={{ display: "flex", gap: 8 }}>
          <button className={`tab${mode === "browse" ? " active" : ""}`} onClick={() => setMode("browse")}>Browse</button>
          <button className={`tab${mode === "study" ? " active" : ""}`} onClick={() => { setMode("study"); setKnown(new Set()); }}>Study Mode</button>
        </div>
      </div>

      {loading ? (
        <div style={{ textAlign: "center", padding: 60 }}>
          <div className="loader" style={{ margin: "0 auto 16px", width: 36, height: 36 }} />
          <div style={{ color: "var(--text2)", fontSize: 14 }}>Generating flashcards…</div>
        </div>
      ) : cards.length > 0 ? (
        <>
          {/* Progress */}
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", fontSize: 13, color: "var(--text2)" }}>
            <span>{current + 1} / {cards.length}</span>
            {mode === "study" && <span style={{ color: "var(--green)" }}>✓ {known.size} known</span>}
          </div>
          <div style={{ height: 4, background: "var(--border)", borderRadius: 99 }}>
            <div style={{ height: "100%", width: `${((current + 1) / cards.length) * 100}%`, background: "var(--accent)", borderRadius: 99, transition: "width .3s" }} />
          </div>

          {/* Card */}
          <div className="flip-card" style={{ height: 280 }} onClick={() => setFlipped(f => !f)}>
            <div className={`flip-card-inner${flipped ? " " : ""}`} style={{ height: "100%", transform: flipped ? "rotateY(180deg)" : "none", transition: "transform .5s cubic-bezier(.22,1,.36,1)", transformStyle: "preserve-3d", position: "relative" }}>
              <div className="flip-card-front card" style={{ height: "100%", display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", padding: 32, textAlign: "center", backfaceVisibility: "hidden" }}>
                <div style={{ fontSize: 11, fontWeight: 700, letterSpacing: 1, color: "var(--text3)", marginBottom: 16, textTransform: "uppercase" }}>Question</div>
                <div style={{ fontSize: 20, fontWeight: 700, lineHeight: 1.5 }}>{cards[current]?.front}</div>
                <div style={{ marginTop: 20, fontSize: 12, color: "var(--text3)" }}>Click to reveal answer</div>
              </div>
              <div className="flip-card-back card" style={{ height: "100%", display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", padding: 32, textAlign: "center", background: "linear-gradient(135deg,#eef0ff,#f3e8ff)", backfaceVisibility: "hidden", position: "absolute", top: 0, left: 0, width: "100%", transform: "rotateY(180deg)" }}>
                <div style={{ fontSize: 11, fontWeight: 700, letterSpacing: 1, color: "var(--accent)", marginBottom: 16, textTransform: "uppercase" }}>Answer</div>
                <div style={{ fontSize: 17, lineHeight: 1.6 }}>{cards[current]?.back}</div>
              </div>
            </div>
          </div>

          {/* Controls */}
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
            <button className="btn btn-ghost" onClick={prev} disabled={current === 0}>← Prev</button>
            <div style={{ display: "flex", gap: 10 }}>
              {mode === "study" && <>
                <button className="btn btn-danger" style={{ fontSize: 13 }} onClick={next}>✗ Still learning</button>
                <button className="btn btn-secondary" style={{ background: "#d1fae5", color: "#059669", fontSize: 13 }} onClick={markKnown}>✓ Got it!</button>
              </>}
              {mode === "browse" && <button className="btn btn-secondary" onClick={() => setFlipped(f => !f)}>Flip card</button>}
            </div>
            <button className="btn btn-ghost" onClick={next} disabled={current === cards.length - 1}>Next →</button>
          </div>

          {/* All cards grid */}
          <div>
            <h3 style={{ fontWeight: 700, fontSize: 14, marginBottom: 12 }}>All cards</h3>
            <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill,minmax(200px,1fr))", gap: 10 }}>
              {cards.map((c, i) => (
                <div key={i} className="card" style={{ padding: 14, cursor: "pointer", border: i === current ? "1.5px solid var(--accent)" : undefined, background: known.has(i) ? "#f0fdf4" : "var(--surface)" }} onClick={() => { setCurrent(i); setFlipped(false); }}>
                  <div style={{ fontSize: 11, fontWeight: 700, color: "var(--text3)", marginBottom: 4 }}>#{i + 1} {known.has(i) ? "✓" : ""}</div>
                  <div style={{ fontSize: 13, fontWeight: 600, overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{c.front}</div>
                </div>
              ))}
            </div>
          </div>
        </>
      ) : (
        <div style={{ textAlign: "center", padding: 60, color: "var(--text2)" }}>No cards generated yet</div>
      )}
    </div>
  );
}

// ─── QUIZ PAGE ────────────────────────────────────────────────────────────────
function QuizPage({ note, user, setPage }) {
  const [questions, setQuestions] = useState([]);
  const [answers, setAnswers] = useState({});
  const [submitted, setSubmitted] = useState(false);
  const [loading, setLoading] = useState(false);
  const [difficulty, setDifficulty] = useState("medium");
  const [score, setScore] = useState(0);

  const generate = useCallback(async (n, diff) => {
    if (!n?.text) return;
    setLoading(true); setSubmitted(false); setAnswers({});
    try {
      const result = await callClaudeJSON(
        "You are a quiz generator. Return JSON only.",
        `Generate 5 quiz questions (${diff} difficulty) from these notes. Mix multiple choice and true/false.
Return: {"questions": [{"id": "q1", "type": "mcq", "question": "...", "options": ["A","B","C","D"], "correct": "A", "explanation": "..."},...]}
For true/false: {"id":"q2","type":"tf","question":"...","correct":"True","explanation":"...","options":["True","False"]}
Notes: ${n.text.slice(0, 2000)}`
      );
      if (result?.questions) setQuestions(result.questions);
    } catch {
      setQuestions([
        { id: "q1", type: "tf", question: "AI can help students study more efficiently.", correct: "True", options: ["True", "False"], explanation: "AI tools like StudySnap can significantly boost studying." }
      ]);
    } finally { setLoading(false); }
  }, []);

  useEffect(() => {
    if (note) generate(note, difficulty);
    else { const saved = getSavedNotes(user?.id || ""); if (saved[0]) generate(saved[0], difficulty); }
  }, [note, user, difficulty, generate]);

  const handleSubmit = () => {
    let s = 0;
    questions.forEach(q => { if (answers[q.id] === q.correct) s++; });
    setScore(s); setSubmitted(true);
  };

  const scorePercent = questions.length ? Math.round((score / questions.length) * 100) : 0;

  if (!note && !getSavedNotes(user?.id || "").length) return (
    <div style={{ padding: 40, textAlign: "center", color: "var(--text2)" }}>
      <div style={{ fontSize: 48, marginBottom: 16 }}>🧠</div>
      <div style={{ fontWeight: 600 }}>No quiz available</div>
      <button className="btn btn-primary" style={{ marginTop: 20 }} onClick={() => setPage("upload")}>Upload notes first</button>
    </div>
  );

  return (
    <div className="slide-up" style={{ maxWidth: 720, margin: "0 auto", padding: "28px 24px", display: "flex", flexDirection: "column", gap: 20 }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", flexWrap: "wrap", gap: 12 }}>
        <div>
          <h2 style={{ fontWeight: 800, fontSize: 22, marginBottom: 4 }}>🧠 Quiz</h2>
          <p style={{ color: "var(--text2)", fontSize: 13 }}>{note?.title || "Practice Quiz"}</p>
        </div>
        <div style={{ display: "flex", gap: 8 }}>
          {["easy","medium","hard"].map(d => (
            <button key={d} className={`tab${difficulty === d ? " active" : ""}`} onClick={() => setDifficulty(d)} style={{ textTransform: "capitalize" }}>{d}</button>
          ))}
        </div>
      </div>

      {loading ? (
        <div style={{ textAlign: "center", padding: 60 }}>
          <div className="loader" style={{ margin: "0 auto 16px", width: 36, height: 36 }} />
          <div style={{ color: "var(--text2)", fontSize: 14 }}>Generating quiz…</div>
        </div>
      ) : submitted ? (
        // Results Screen
        <div className="card scale-in" style={{ padding: 40, textAlign: "center" }}>
          <div style={{ fontSize: 56, marginBottom: 16 }}>{scorePercent >= 80 ? "🎉" : scorePercent >= 60 ? "👍" : "📚"}</div>
          <div style={{ fontWeight: 800, fontSize: 28, marginBottom: 4 }}>{scorePercent}%</div>
          <div style={{ color: "var(--text2)", marginBottom: 8 }}>{score} out of {questions.length} correct</div>
          <div style={{ color: scorePercent >= 80 ? "var(--green)" : scorePercent >= 60 ? "var(--amber)" : "var(--red)", fontWeight: 600, marginBottom: 28 }}>
            {scorePercent >= 80 ? "Excellent! You're well prepared." : scorePercent >= 60 ? "Good effort! Review the missed questions." : "Keep studying — you'll get there!"}
          </div>
          <div style={{ display: "flex", gap: 10, justifyContent: "center", flexWrap: "wrap" }}>
            <button className="btn btn-ghost" onClick={() => { setSubmitted(false); setAnswers({}); }}>📋 Review Answers</button>
            <button className="btn btn-gradient" onClick={() => generate(note, difficulty)}>🔄 New Quiz</button>
          </div>
        </div>
      ) : (
        <>
          <div style={{ display: "flex", flexDirection: "column", gap: 16 }}>
            {questions.map((q, i) => (
              <div key={q.id} className="card" style={{ padding: 20, border: submitted && answers[q.id] ? `1.5px solid ${answers[q.id] === q.correct ? "var(--green)" : "var(--red)"}` : undefined }}>
                <div style={{ fontWeight: 700, fontSize: 14, marginBottom: 14 }}>
                  <span className="badge badge-blue" style={{ marginRight: 8, fontSize: 10 }}>Q{i + 1}</span>
                  {q.question}
                </div>
                <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
                  {q.options?.map(opt => {
                    const isSelected = answers[q.id] === opt;
                    const isCorrect = submitted && opt === q.correct;
                    const isWrong = submitted && isSelected && opt !== q.correct;
                    return (
                      <button key={opt} onClick={() => !submitted && setAnswers(a => ({ ...a, [q.id]: opt }))}
                        style={{ display: "flex", gap: 10, alignItems: "center", padding: "10px 14px", borderRadius: 10, border: `1.5px solid ${isCorrect ? "var(--green)" : isWrong ? "var(--red)" : isSelected ? "var(--accent)" : "var(--border)"}`, background: isCorrect ? "#f0fdf4" : isWrong ? "#fee2e2" : isSelected ? "var(--accent-light)" : "var(--surface)", cursor: submitted ? "default" : "pointer", fontFamily: "Sora,sans-serif", fontSize: 13, fontWeight: isSelected ? 600 : 400, textAlign: "left", transition: "all .15s" }}>
                        <span style={{ width: 22, height: 22, borderRadius: "50%", background: isCorrect ? "var(--green)" : isWrong ? "var(--red)" : isSelected ? "var(--accent)" : "var(--border)", display: "flex", alignItems: "center", justifyContent: "center", color: (isCorrect || isWrong || isSelected) ? "#fff" : "var(--text3)", fontSize: 11, fontWeight: 700, flexShrink: 0 }}>
                          {isCorrect ? "✓" : isWrong ? "✗" : opt[0]}
                        </span>
                        {opt}
                      </button>
                    );
                  })}
                </div>
                {submitted && q.explanation && (
                  <div style={{ marginTop: 12, padding: "10px 14px", background: "var(--accent-light)", borderRadius: 8, fontSize: 12, color: "var(--accent)", lineHeight: 1.6 }}>
                    💡 {q.explanation}
                  </div>
                )}
              </div>
            ))}
          </div>

          {questions.length > 0 && (
            <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
              <div style={{ fontSize: 13, color: "var(--text2)" }}>{Object.keys(answers).length}/{questions.length} answered</div>
              <button className="btn btn-gradient" style={{ padding: "12px 28px" }} onClick={handleSubmit} disabled={Object.keys(answers).length < questions.length}>
                Submit Quiz →
              </button>
            </div>
          )}
        </>
      )}
    </div>
  );
}

// ─── SAVED NOTES PAGE ─────────────────────────────────────────────────────────
function NotesPage({ user, setPage, setEditNote }) {
  const [notes, setNotes] = useState([]);
  const [search, setSearch] = useState("");
  const [filterSubject, setFilterSubject] = useState("all");

  useEffect(() => { if (user?.id) setNotes(getSavedNotes(user.id)); }, [user]);

  const filtered = notes.filter(n => {
    const matchSearch = !search || (n.title || "").toLowerCase().includes(search.toLowerCase()) || (n.summary || "").toLowerCase().includes(search.toLowerCase());
    const matchSubject = filterSubject === "all" || n.subject === filterSubject;
    return matchSearch && matchSubject;
  });

  const handleDelete = (id) => {
    deleteNote(user.id, id);
    setNotes(getSavedNotes(user.id));
  };

  return (
    <div className="slide-up" style={{ maxWidth: 900, margin: "0 auto", padding: "28px 24px", display: "flex", flexDirection: "column", gap: 20 }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", flexWrap: "wrap", gap: 12 }}>
        <div>
          <h2 style={{ fontWeight: 800, fontSize: 22, marginBottom: 4 }}>📁 Saved Notes</h2>
          <p style={{ color: "var(--text2)", fontSize: 13 }}>{notes.length} saved notes</p>
        </div>
        <button className="btn btn-gradient" onClick={() => setPage("upload")}>⚡ New Note</button>
      </div>

      <div style={{ display: "flex", gap: 12, flexWrap: "wrap" }}>
        <input placeholder="🔍 Search notes…" value={search} onChange={e => setSearch(e.target.value)} style={{ flex: 1, minWidth: 200 }} />
        <select value={filterSubject} onChange={e => setFilterSubject(e.target.value)} style={{ width: "auto" }}>
          <option value="all">All subjects</option>
          {SUBJECTS.map(s => <option key={s.id} value={s.id}>{s.icon} {s.label}</option>)}
        </select>
      </div>

      {filtered.length === 0 ? (
        <div style={{ textAlign: "center", padding: 60, color: "var(--text2)" }}>
          <div style={{ fontSize: 48, marginBottom: 16 }}>📭</div>
          <div style={{ fontWeight: 600 }}>{search ? "No notes match your search" : "No saved notes yet"}</div>
          {!search && <button className="btn btn-primary" style={{ marginTop: 20 }} onClick={() => setPage("upload")}>Process your first note</button>}
        </div>
      ) : (
        <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill,minmax(280px,1fr))", gap: 16 }}>
          {filtered.map(note => (
            <div key={note.id} className="card card-hover" style={{ padding: 20, position: "relative" }}>
              <div style={{ display: "flex", gap: 8, alignItems: "center", marginBottom: 10 }}>
                <span style={{ fontSize: 18 }}>{SUBJECTS.find(s => s.id === note.subject)?.icon || "📝"}</span>
                <span className="badge badge-blue" style={{ fontSize: 10 }}>{note.subject}</span>
                <span style={{ marginLeft: "auto", fontSize: 11, color: "var(--text3)" }}>{new Date(note.savedAt).toLocaleDateString()}</span>
              </div>
              <div style={{ fontWeight: 700, fontSize: 15, marginBottom: 6 }}>{note.title || "Untitled"}</div>
              <div style={{ fontSize: 12, color: "var(--text2)", lineHeight: 1.6, overflow: "hidden", display: "-webkit-box", WebkitLineClamp: 3, WebkitBoxOrient: "vertical", marginBottom: 14 }}>
                {note.summary || note.text?.slice(0, 120)}
              </div>
              <div style={{ display: "flex", gap: 8 }}>
                <button className="btn btn-secondary" style={{ fontSize: 12, padding: "6px 12px", flex: 1 }} onClick={() => { setEditNote(note); setPage("results"); }}>View</button>
                <button className="btn btn-danger" style={{ fontSize: 12, padding: "6px 12px" }} onClick={() => handleDelete(note.id)}>🗑</button>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

                <td style={{ padding: "10px 0", textAlign: "center", color: free === "✗" ? "var(--text3)" : "var(--text2)" }}>{free}</td>
                <td style={{ padding: "10px 0", textAlign: "center", color: "var(--accent)", fontWeight: 600 }}>{prem}</td>

            ))}
          </tbody>
        </table>
      </div>

      <div style={{ fontSize: 13, color: "var(--text3)" }}>💳 Secure payments · Cancel anytime · No hidden fees</div>
    </div>
  );

// ─── MAIN APP ─────────────────────────────────────────────────────────────────
export default function App() {
  const auth = useAuth();
  const [authMode, setAuthMode] = useState("login");
  const [appMode, setAppMode] = useState("landing"); // landing | auth | app
  const [page, setPage] = useState("dashboard");
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [currentNote, setCurrentNote] = useState(null);
  const [editNote, setEditNote] = useState(null);
  const [flashcardNote, setFlashcardNote] = useState(null);
  const [quizNote, setQuizNote] = useState(null);

  useEffect(() => {
    if (auth.user) setAppMode("app");
  }, [auth.user]);

  const handleLogin = async (name, email, password) => {
    const u = auth.login(email, password);
    setAppMode("app");
    return u;
  };

  const handleSignup = async (name, email, password) => {
    if (!name.trim()) throw new Error("Please enter your name");
    const u = auth.signup(name, email, password);
    setAppMode("app");
    return u;
  };

  const handleLogout = () => {
    auth.logout();
    setAppMode("landing");
    setPage("dashboard");
    setCurrentNote(null);
  };

  const handleProcess = async (text, title, subject) => {
    const noteId = crypto.randomUUID();
    const baseNote = { id: noteId, text, title, subject, savedAt: new Date().toISOString() };
    setCurrentNote(baseNote);
    setEditNote(null);
    setPage("results");
    auth.recordUsage();

    try {
      const [summary, bullets, simple, examPoints, nightBefore, questions, keywords] = await Promise.all([
        callClaude("You are a study assistant. Be concise.", `Write a clear 3-4 sentence summary of these notes:\n${text.slice(0, 3000)}`),
        callClaude("You are a study assistant. Format as bullet points.", `Create 6-8 bullet point key facts from these notes:\n${text.slice(0, 3000)}`),
        callClaude("You are a study assistant. Explain simply.", `Explain these notes in simple, easy-to-understand language for a student:\n${text.slice(0, 3000)}`),
        callClaude("You are an exam coach.", `List the 5-7 most important exam-ready points from these notes. Format as numbered list:\n${text.slice(0, 3000)}`),
        callClaude("You are an exam coach. Be ultra-brief.", `Create a VERY SHORT night-before-exam revision summary (max 150 words) with only the most critical points:\n${text.slice(0, 3000)}`),
        callClaude("You are an exam coach.", `Generate 5 likely exam questions based on these notes. Format as numbered list:\n${text.slice(0, 3000)}`),
        callClaude("You are a study assistant.", `Extract 8-10 key terms, definitions, and formulas from these notes. Format each as "Term: Definition":\n${text.slice(0, 3000)}`),
      ]);
      const fullNote = { ...baseNote, summary, bullets, simple, examPoints, nightBefore, questions, keywords };
      setCurrentNote(fullNote);
    } catch (e) {
      setCurrentNote(prev => ({ ...prev, summary: "Error generating summary. Please try again.", bullets: "", simple: "", examPoints: "", nightBefore: "", questions: "", keywords: "" }));
    }
  };

  const handleSaveNote = (note) => {
    if (auth.user?.id) saveNote(auth.user.id, note);
  };

  // Page title map
  const titles = { dashboard: "Dashboard", upload: "New Study Session", results: "Study Pack", flashcards: "Flashcards", quiz: "Quiz", notes: "Saved Notes", pricing: "Upgrade", settings: "Settings" };

  // ── Routing ──
  if (appMode === "landing") return (
    <>
      <GlobalStyle />
      <LandingPage onLogin={() => { setAuthMode("login"); setAppMode("auth"); }} onSignup={() => { setAuthMode("signup"); setAppMode("auth"); }} />
    </>
  );

  if (appMode === "auth") return (
    <>
      <GlobalStyle />
      <AuthPage mode={authMode} onAuth={authMode === "login" ? handleLogin : handleSignup} onSwitch={() => setAuthMode(m => m === "login" ? "signup" : "login")} />
    </>
  );

  // ── App ──
  const activeNote = editNote || currentNote;

  return (
    <>
      <GlobalStyle />
      <div style={{ display: "flex", minHeight: "100vh" }}>
        <Sidebar page={page} setPage={setPage} user={auth.user} onLogout={handleLogout} isOpen={sidebarOpen} onClose={() => setSidebarOpen(false)} />

        <div className="main-content" style={{ flex: 1, marginLeft: 240, display: "flex", flexDirection: "column", minHeight: "100vh", minWidth: 0 }}>
          <TopBar title={titles[page] || "StudySnap"} onMenuToggle={() => setSidebarOpen(o => !o)} user={auth.user} />

          <main style={{ flex: 1, overflowY: "auto" }}>
            {page === "dashboard" && <Dashboard user={auth.user} setPage={setPage} setEditNote={setEditNote} />}
            {page === "upload" && <UploadPage user={auth.user} onProcess={handleProcess} canUse={auth.canUseAI} setPage={setPage} />}
            {page === "results" && <ResultsPage note={activeNote} user={auth.user} onSave={handleSaveNote} setPage={setPage} setFlashcardNote={setFlashcardNote} setQuizNote={setQuizNote} />}
            {page === "flashcards" && <FlashcardsPage note={flashcardNote || activeNote} user={auth.user} setPage={setPage} />}
            {page === "quiz" && <QuizPage note={quizNote || activeNote} user={auth.user} setPage={setPage} />}
            {page === "notes" && <NotesPage user={auth.user} setPage={setPage} setEditNote={setEditNote} />}
            {page === "pricing" && <PricingPage user={auth.user} updateUser={auth.updateUser} />}
            {page === "settings" && <SettingsPage user={auth.user} updateUser={auth.updateUser} onLogout={handleLogout} />}
          </main>
        </div>
      </div>
    </>
  );
}
